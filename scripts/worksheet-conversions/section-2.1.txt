\begin{parts}

\begin{solution}[3em]
The slope is $-7$, the vertical intercept is $9$.
\end{solution}

The slope is $\dfrac{1}{3}$, the vertical intercept is $0$.

The slope is $0$, the vertical intercept is $3.6$.

\question The rental charge, $C$, at a video store is given by $C=4.29+3.99n$, where $n$ is the number of days greater than $2$ for which the video is kept. Give the initial value and the slope and explain their meaning in practical terms.

\begin{solution}[10em]
The initial value (vertical intercept) is $4.29$. Therefore, the cost to rent from the video store for only $2$ days is $\$4.29$.

The slope is $3.99$. Therefore, the cost to rent from the video store goes up by $\$3.99$ for each day over $2$ days.
\end{solution}



\question The number of people, $P$, remaining in the lecture hall $s$ seconds after the MTH 103 lecture has ended is given by $P=75-5s$. Give the initial value and the slope and explain their meaning in practical terms.

\begin{solution}[10em]
The initial value (vertical intercept) is $75$. Thus, there are $75$ people attending lecture.

The slope is $-5$. Thus, the number of people remaining in the lecture hall goes down by $5$ people for each second after the end of lecture.
\end{solution}



\question Find a formula for the linear function $f(x)$ if $f(-10)=15$ and $f(15)=-20$.

\begin{solution}[12em]
Use the ordered pairs $(-10,15)$ and $(15,-20)$. Then
\[
m = \frac{-20-15}{15-(-10)}  = \frac{-35}{25}  = -\frac{7}{5}
.\] Plugging into $f(x)=mx+b$, then
\[
f(x) = -\frac{7}{5}x + b
.\] Using any ordered pair, say $(-10,15)$, then
\[
15=-\frac{7}{5} (-10)  + b \quad \implies \quad 15=14+b \quad \implies \quad b =1
.\] Therefore $f(x) = -\dfrac{7}{5}x+1$.
\end{solution}

\question Find a formula for the linear function $g(x)$ if $g(2)=6$ and $g(4)=14$.

\begin{solution}[12em]
Use the ordered pairs $(2,6)$ and $(4,14)$. Then
\[
m = \frac{14-6}{4-2}  = \frac{8}{2}=4
.\] Plugging into $g(x)=mx+b$, then
\[
g(x) = 4x+b
.\] Using any ordered pair, say $(2,6)$, then
\[
6=4(2) + b \quad \implies \quad 6 = 8 + b \quad \implies \quad b = -2
.\] Therefore $g(x)=4x-2$.
\end{solution}


\question Find an equation of the line passing through $(-1,5)$ and $(0,5)$.

\begin{solution}[12em]
Note that $(0,5)$ gives us the vertical intercept already, so $b=5$. The slope is
\[
m = \frac{5-5}{0-(-1)} = \frac{0}{1}=0
.\] Therefore the equation is $y=0x + 5$ or simply $y=5$.
\end{solution}
