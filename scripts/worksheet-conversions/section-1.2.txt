
\question Create a table of values and sketch the graph of the function $y=x^3$ for $-2\leq x\leq 2$. Then use your calculator or any other graphing utility to check your graph.


\begin{solution}[16em]
\begin{minipage}{0.4\textwidth}
\centering
\begin{tabular}{c|r}
$x$ & $y=x^3$ \\\hline
$-2$ & $(-2)^3=-8$\\\hline
$-1$ & $(-1)^3=-1$\\\hline
$0$ & $0^3=0$\\\hline
$1$ & $1^3 = 1$\\\hline
$2$ & $2^3 = 8$
\end{tabular} 
\end{minipage} \hfill
\begin{minipage}{0.6\textwidth}
\centering
\begin{tikzpicture}[scale=0.8]
\begin{axis}[grid=both,
axis lines=middle,
xmin=-3,xmax=3,
ymin=-9,ymax=9,
xtick={-2,-1,1,2},
ytick={-8,-6,-4,-2,2,4,6,8},
minor tick={},
xlabel=\(x\), ylabel=\(y\),
samples=250]
\addplot[domain=-2:2,thick] {x^3};
\end{axis}
\end{tikzpicture}
\end{minipage} 


\end{solution}

\question The amount of caffeine remaining in the body, $C$, in milligrams, $t$ hours after drinking a cup of coffee, $C=C(t)$ is given by the graph below:

\begin{center}
\begin{tikzpicture}[scale=0.8]
\begin{axis}[grid=both,
axis lines=middle,
axis line style = very thick,
xmin=0,xmax=20,
ymin=0,ymax=96,
xtick={5,10,15,20},
ytick={24,48,72,96},
minor tick={},
xlabel=\(t\), ylabel=\(C\),
samples=250]
\addplot[domain=0:20,thick] {96*0.5^(x/5)};
\end{axis}
\end{tikzpicture}
\end{center}

\begin{parts}
\part How much caffeine was absorbed into the bloodstream from the cup of coffee?

\begin{solution}[4em]
This is the value at $t=0$ hours, so $96$ milligrams.
\end{solution}

\part How much caffeine is left after 5 hours? After 10 hours?

\begin{solution}[4em]
After $5$ hours, there are $C(5)=48$ mg of caffeine remaining. After $10$ hours, there are $C(10)=24$ mg of caffeine remaining.
\end{solution}


\part Is the function $C(t)$ increasing, decreasing or neither on the interval $t\geq 0$?

\begin{solution}[4em]
The function is decreasing on $t\ge 0$; the outputs decrease as the inputs increase.
\end{solution}

\end{parts}

\question For the function $y=f(t)$ whose graph is depicted below, identify intervals on the $t$-axis on which the function is increasing and the intervals on the $t$-axis on which the function is decreasing.

\begin{center}
\begin{tikzpicture}[scale=0.9]
\begin{axis}[grid=both,
axis lines=middle,
axis line style = very thick,
xmin=0,xmax=13,
ymin=-2,ymax=4,
xtick={1,...,12},
ytick={-1,1,2,3},
minor tick={},
xlabel=\(t\), ylabel=\(y\),
trig format = rad,
samples=250]
\addplot[domain=0:12,thick] {2*sin(2*3.14/12*x)+1};
\end{axis}
\end{tikzpicture}
\end{center}

\begin{solution}
The function is increasing on the intervals $(0,3)$ and $(9,12)$, which in inequality notation reads $0<t<3$ and $9<t<12$, respectively.

The function is decreasing on the interval $(3,9)$, which in inequality notation reads $3<t<9$.
\end{solution}
