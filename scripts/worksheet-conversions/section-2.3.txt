\question Decide if the given table could represent a linear function and explain your reasoning.  For each table that could represent a linear function, find a formula for that function.

\begin{parts}
\part

\begin{tabular}{|c|c|c|c|c|} \hline
$x$ & $0$ & $1$ & $2$ & $3$ \\\hline
$y$ & $12.6$ & $9.3$ & $6.0$ & $2.7$ \\\hline
\end{tabular}

\begin{solution}[8em]
Notice that $x$ changes by $+1$ moving left to right, and $y$ changes by $-3.3$. Therefore, the slope between each pair of consecutive points is always $\frac{-3.3}{1} =-3.3$, and because this is the same number for all, this is a linear function.

Its slope is $-3.3$ and its vertical intercept is $12.6$ (from the table) so
\[
y = -3.3x + 12.6
.\] 
\end{solution}



\part

\begin{tabular}{|c|c|c|c|c|c|} \hline
$t$ & $-1$ & $1$ & $3$ & $5$ \\\hline
$f(t)$ & $0.1$ & $0.4$ & $0.7$ & $1.0$ \\\hline
\end{tabular}

\begin{solution}[8em]
The change in $t$ (the ``run'') is always $2$, and the change in $f(t)$ (the ``rise'') is always $0.3$, and so this is a linear function with slope $m=\frac{0.3}{2} = 0.15$.

To find the equation, either use point--slope form which would look something like
\[
f(t)=0.15(t-(-1))+0.1 = 0.15(t+1)+0.1
\] or use slope--intercept form and solve for $b$. That is,
\[
f(-1) = 0.1 = 0.15(-1) + b \quad \implies \quad b = 0.25
\] and so $f(t)=0.15t + 0.25$.
\end{solution}



\part

\begin{tabular}{|c|c|c|c|c|c|c|}\hline
$n$ & $0$ & $1.5$ & $3$ & $4.5$ & $6$  \\\hline
$g(n)$ & $12$ & $18$ & $12$ & $6$ & $12$ \\\hline
\end{tabular}

\begin{solution}[8em]
The change in $n$ (the ``run'') is always $1.5$, but the change in $g(n)$ (the ``rise'') changes --- it is $6$, $-6$, $-6$, and then $6$. Because these are not always the same number, the slopes between consecutive points are not equal, and so the function is nonlinear.
\end{solution}


\part

\begin{tabular}{|c|c|c|c|c|c|c|} \hline
$t$ & $0$ & $2$ & $4$ & $6$ & $8$ \\\hline
$h(t)$ & $6.4$ & $1.6$ & $0.4$ & $0.1$ & $0.025$ \\\hline
\end{tabular}

\begin{solution}[8em]
The change in $t$ (the ``run'') is always $2$, but the change in $h(t)$ (the ``rise'') changes --- it is $-4.4$, $-1.2$, $-0.3$, and then $-0.0075$. Because these are not always the same number, the slopes between consecutive points are not equal, and so the function is nonlinear.
\end{solution}


\part

\begin{tabular}{|c|c|c|c|c|c|}\hline
$z$ & $-2$ & $1$ & $4$ & $7$ \\\hline
$w(z)$ & $6$ & $4.8$ & $3.6$ & $2.4$ \\\hline
\end{tabular}

\begin{solution}
The change in $z$ is always $3$, and the change in $w(z)$ is always $-1.2$ and so this is a linear function. The slope is $m = \frac{-1.2}{3} = -0.4$, and we can find the equation using point--slope form such as
\[
w(z) = -0.4(z-(-2)) + 6 = -0.4(z+2)+6
\] or use slope--intercept form and solve for $b$, such as
\[
w(z) = -0.4z + b \quad \implies \quad 6 = -0.4(-2) + b \quad \implies \quad b = 5.2
.\] And therefore $w(z) = -0.4z + 5.2$.  
\end{solution}


\end{parts}

\question The height of a human individual can be estimated by the length of the femur, as shown for males in the following table.

\begin{center}
\begin{tabular}{|c|c|c|c|c|}\hline
$L$ (length of femur in cm) & $42$ & $45$ & $48$  & $51$ \\\hline
$H$ (height of human male in cm) & $162.97$ & $169.93$ & $176.89$ & $183.85$ \\\hline
\end{tabular}
\end{center}

\begin{parts}
\part Use the information in this table to find a possible formula for the height of a human male $H$ as a function of his femur length $L$.

\begin{solution}[14em]
Note that the change in $L$ is $3$, and the change in $H$ is always $6.96$ and so this is a linear function with slope $m = \frac{6.96}{3}  = 2.32$. Now, either use point--slope form or slope--intercept form to obtain the equation for the function. Using slope--intercept form, then
\[
H = 2.32L + b
.\] Plugging in any point, say $(42,162.97)$, results in
\[
162.97 = 2.32 (42) + b
\] meaning that $b = 65.53$ and so
\[
H = 2.32L + 65.53
.\]  
\end{solution}


\part What would the approximate height of a human male with femur length $46$ cm be?

\begin{solution}[7em]
By plugging in $L = 46$, then the approximate height is
\[
H = 2.32(46) + 65.53 = 172.23 \text{ cm} 
.\] 
\end{solution}


\part Fernando is $175.4$ cm tall. What would you expect the approximate length of his femur to be?

\begin{solution}
With a height of $H = 175.4$, then
\[
175.4 = 2.32L + 65.53
.\] Solving for $L$, we get
\[
109.87 = 2.32 L \quad \implies \quad L = 47.36 \text{ cm} 
.\] 
\end{solution}

\end{parts}
