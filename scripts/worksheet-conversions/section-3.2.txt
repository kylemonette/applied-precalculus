\fullwidth{
\noindent Find the horizontal intercepts of each quadratic function by factoring.
}

\begin{questions}

\question $f(x)=x^2+3 x+2$

\begin{solution}[6em]
Two numbers that multiply to $2$ and add to $3$ are $1$ and $2$, so $f(x)=(x+2)(x+1)$. Setting this to $0$, then $x+2=0$ and $x+1=0$, meaning $x=-2$ and $x=-1$ are the horizontal intercepts.
\end{solution}


\question $y=x^2-x-6$

\begin{solution}[6em]
Two numbers that multiply to $-6$ and add to $-1$ are $-3$ and $2$, so $y=(x-3)(x+2)$. Setting this to $0$, then $x-3=0$ and $x+2=0$, meaning $x=3$ and $x=-2$ are the horizontal intercepts.
\end{solution}


\question $g(t)=t^2+t-6$

\begin{solution}[6em]
Two numbers that multiply to $-6$ and add to $1$ are $3$ and $-2$, so $g(t)=(t+3)(t-2)$. Setting this to $0$, then $t+3=0$ and $t-2=0$, meaning $t=-3$ and $t=2$ are the horizontal intercepts.
\end{solution}


\question $y=x^2-8 x+16$

\begin{solution}[6em]
Two numbers that multiply to $16$ and add to $-8$ are $-4$ and $-4$, so $y=(x-4)(x-4)$. Setting this to $0$, then $x-4=0$ and so $x=4$ is the only horizontal intercept.
\end{solution}


\question $f(w)=w^2+14 w+49$

\begin{solution}[6em]
Two numbers that multiply to $49$ and add to $14$ are $7$ and $7$, so $f(w)=(w+7)(w+7)$. Setting this to $0$, then $w+7=0$ and so $w=-7$ is the only horizontal intercept.
\end{solution}


\question $f(t)=64t^2-36$

\begin{solution}[6em]
One method (the preferred method) is to realize this is a {\it difference of squares}. That is, $f(t)=(8t)^2-6^2$ and so $f(t)=(8t+6)(8t-6)$. In this case, setting $f(t)=0$ yields $8t-6=0$ so $t=\frac{6}{8}=\frac{3}{4}$ and also $8t+6=0$ so $t=-\frac{3}{4}$.

Another method is to factor out $64$ and write $f(t)=64(t^2-9 / 16)$. Then, we need two numbers that multiply to $-9/16$ and add to $0$, which are $-3/4$ and $3 /4$. Thus $f(t)=64(t-3/4)(t+3/4)$ which is actually equal to the other method above (as we expect) if you distribute the $64$ inside.
\end{solution}


\question $h(x)=2 x^2+10 x+8$

\begin{solution}[15em]
Note that $2$ is a factor in each term, so we can write $h(x) = 2 (x^2+5x+4)$. Now, concentrate on $x^2+5x+4$. Two numbers that multiply to $4$ and add to $5$ are $4$ and $1$, so this factors to $(x+4)(x+1)$. Therefore $h(x)=2(x+4)(x+1)$. Horizontal intercepts are $x=-4$ and $x=-1$.
\end{solution}


\question $y=2 w^2- w-3$

\begin{solution}[12em]
Unlike the last question, $2$ is not a factor in every term. So we use the $ac$-method (described in the textbook). First, find two numbers which multiply to $a\cdot c = 2 \cdot (-3)=-6$ and add to $-1$, which are $2$ and $-3$. Then, write $y = 2w^2 +2w - 3w - 3$. Factoring out the greatest common factors,
\[
y = 2w(w+1) - 3\cdot (w+1) 
\] which factors to $y=(w+1)(2w-3)$. Setting this to $0$, then $w+1=0$ and $2w-3=0$. Thus horizontal intercepts are $w=-1$ and $w=\frac{3}{2}$. 
\end{solution}


\question $f(x)=-x^2-6x+16$

\begin{solution}[6em]
Factor out the $-1$ from everything, yielding $f(x)=-(x^2+6x-16)$. Then, concentrate on $x^2+6x-16$. Two numbers which multiply to $-16$ and add to $6$ are $8$ and $-2$. Therefore $f(x)=-(x+8)(x-2)$. Setting this to $0$, then $x+8=0$ and $x-2=0$, thus $x=-8$ and $x=2$ are the horizontal intercepts.
\end{solution}


\question $g(x)=\frac{1}{49}-4x^2$

\begin{solution}[6em]
This is another {\it difference of squares} since $g(x)=\pbr{\frac{1}{7}} ^2 - (2x)^2$. Therefore $g(x)=\pbr{\frac{1}{7}-2x} \pbr{\frac{1}{7}+2x} $. This means that the horizontal intercepts are when $\frac{1}{7}-2x=0$ and $\frac{1}{7}+2x=0$, that is, when $x=\frac{1}{14}$ and $x=-\frac{1}{14}$.
\end{solution}


\question $p(x)=2x^2-6x$

\begin{solution}[6em]
The factor $2x$ is common to both terms, so $p(x)=2x(x-3)$ and therefore the horizontal intercepts are when $x=0$ and $x=3$.

{\it Alternatively, you can just factor out the $2$ and then write $p(x)=2(x^2-3x)$. From here, find two numbers that multiply to $0$ and add to $-3$. This will yield the same answer as the above.}
\end{solution}


\question $m(t)=-5t^2+2t$

\begin{solution}[6em]
The greatest common factor in each term is $t$, so $m(t)=t(-5t+2)$. Then, horizontal intercepts are when $t=0$ and when $-5t+2=0$, that is, $t=0$ and $t=\frac{2}{5}$.
\end{solution}


\end{questions}
