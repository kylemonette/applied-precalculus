\question Determine whether each equation has one solution, no solutions, or an infinite number of solutions. If there is a solution, find it.

\begin{parts}
\part $3(x-1)=-3(x+1)$

\begin{solution}[10em]
Distributing on each side,
\[
3x-3 = -3x-3
.\] Adding $3$ to both sides,
\[
3x=-3x
.\] Adding $3x$ to both sides,
\[
6x=0
.\] Therefore $x=0$ is the only solution.
\end{solution}


\part $6x+5=-4$

\begin{solution}[10em]
Subtracting $5$ from both sides,
\[
6x=-9
.\] Dividing both sides by $6$,
\[
x = -\frac{9}{6} = -\frac{3}{2}
\] is the only solution.
\end{solution}


\part $5x+7=5(x+1)+1$

\begin{solution}[10em]
Distributing on the right side,
\[
5x+7 = 5x+5+1 
.\] Therefore we have
\[
5x+7 = 5x+6
.\] Subtracting $5x$ from both sides,
\[
7=6
.\] Clearly impossible!! So there is no solution.
\end{solution}

\part $-2x+3=-2(x-1)+1$

\begin{solution}[10em]
Distributing on the right side,
\[
-2x+3 = -2x+2+1
.\] Therefore we have
\[
-2x +3 = -2x+3
.\] Adding $2x$ to both sides,
\[
3=3
.\] Clearly always true!! Therefore there are infinitely many solutions.
\end{solution}


\end{parts}

\question Solve the equation $g(t)=5$ where $g(t)=\dfrac{-7}{6-2t}$.

\begin{solution}[15em]
We set up
\[
5 = \frac{-7}{6-2t}
.\] Multiplying both sides by $6-2t$, then
\[
5(6-2t) = -7
.\] Distributing on the left,
\[
30 - 10t = -7
\] and then subtracting $30$ from both sides,
\[
-10t = -37 \quad \implies \quad t = \frac{-37}{-10}  = 3.7
.\] 
\end{solution}


\question Is each line horizontal, vertical, or neither? If it is horizontal or vertical, write an equation for the line and graph the line.

\begin{parts}
\part The line through $(0,6)$ and $(1,6)$.

\begin{solution}[10em]
The slope is given by
\[
m = \frac{6-6}{1-0}  = \frac{0}{1}=0
.\] Therefore the line is horizontal, and further the vertical intercept was given as $b=6$ so $y=6$ is the equation. The graph is below.

\begin{center}
\begin{tikzpicture}[scale=0.8]
\begin{axis}[grid=none,
axis lines=middle,
axis line style = very thick,
xmin=-8,xmax=8,
ymin=-8,ymax=8,
xtick={-8,-6,...,8},
ytick={-8,-6,...,8},
minor tick={},
xlabel=\(x\), ylabel=\(y\),
samples=250]
\addplot[domain=-7:7,thick] {6};
\end{axis}
\end{tikzpicture}
\end{center} 

\end{solution}


\part The line through $(1,3)$ and $(-4,1)$.

\begin{solution}[10em]
The slope is 
\[
m = \frac{1-3}{-4-1}  = \frac{-2}{-5}  = \frac{2}{5}
.\] So the line is neither vertical nor horizontal.
\end{solution}


\part The line through $(2,-5)$ and $(2,4)$.

\begin{solution}[10em]
The slope is
\[
m = \frac{4-(-5)}{2-2}  = \frac{9}{0} \qquad \text{which is undefined!} 
\] Therefore the line is vertical, and the $x$-intercept is $x=2$. The graph is below.

\begin{center}
\begin{tikzpicture}[scale=0.8]
\begin{axis}[grid=none,
axis lines=middle,
axis line style = very thick,
xmin=-3,xmax=3,
ymin=-3,ymax=3,
xtick={-3,-2,-1,1,2,3},
xticklabels={-3,-2,-1,1,2,3},
ytick={-3,-2,-1,1,2,3},
minor tick={},
xlabel=\(x\), ylabel=\(y\),
samples=250]
\draw[thick] (axis cs:2,3)--(axis cs:2,-3);
\end{axis}
\end{tikzpicture}
\end{center} 

\end{solution}

\end{parts}
