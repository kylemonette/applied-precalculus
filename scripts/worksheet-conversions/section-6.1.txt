\question Evaluate without using a calculator, or say that the expression is undefined.

\begin{parts}
\part $\ln (e^9)$

\begin{solution}[4em]
If $\ln (e^9)=x$ then $e^x = e^9$, so $x$ must be $9$. Hence $\ln (e^9)=9$.
\end{solution}

\part $\log_4 (\sqrt{4^3})$

\begin{solution}[4em]
If $\log_4(\sqrt{4^3} )=x$ then $4^x = \sqrt{4^3} $. Note that $\sqrt{4^3} = (4^3)^{1 /2} = 4^{3 /2}$. Therefore $4^x = 4^{3 /2}$ and thus $x=\frac{3}{2}$ is the value of the logarithm.
\end{solution}

\part $\log (0.0001)$

\begin{solution}[4em]
{\it Recall: $\log $ with no subscript means base $10$}

Because $0.0001 = 10^{-4}$, then this is $\log (10^{-4})$. If $\log (10^{-4})=x$, then $10^x = 10^{-4}$. Therefore $x=-4$ and so $\log (0.0001)=-4$.
\end{solution}

\part $\log_2(-8)$

\begin{solution}[4em]
This is not defined. Recall that $\log _b(x)$ is only defined for $x>0$.
\end{solution}

\part $\log_7 \pbr{\dfrac{1}{343}}$

\begin{solution}[4em]
Note that $343 = 7^3$ so that $\frac{1}{343}=7^{-3}$. Therefore $\log_7(7^{-3})=x$ means that $7^x = 7^{-3}$ and so $x=-3$, implying that the logarithm value is $-3$.
\end{solution}

\part $\log_4(16)$

\begin{solution}[4em]
Because $16=4^2$, then we have $\log _4(4^2)$. If $\log _4(4^2)=x$, then $4^x=4^2$ and so $x=2$. Therefore $\log _4(16)=2$.
\end{solution}

\part $\ln \pbr{\dfrac{e^2}{\sqrt[3]{e}}}$

\begin{solution}[5em]
We can simplify the fraction here to be
\[
\frac{e^2}{\sqrt[3]{e} } \quad = \quad \frac{e^2}{e^{1 /3}}  \quad = \quad e^{2-1/3} \quad = \quad e^{5/3} 
.\] Now, we are computing $\ln (e^{5 /3})$. If $\ln (e^{5 /3})=x$, then $e^x = e^{5 /3}$ so $x=\frac{5}{3}$ is the logarithm value.
\end{solution}

\part $\log_9 \pbr{\dfrac{1}{6561}}$

\begin{solution}[4em]
Note that $6561 = 9^4$ and so $\frac{1}{6561} = 9^{-4}$. Therefore we have $\log _9(9^{-4})$ and this is just $-4$.
\end{solution}

\end{parts}


\question Use the definition of the logarithm to solve each equation.

\begin{parts}
\part $2^x=63$

\begin{solution}[5em]
With the definition of the logarithm, then $\log _2(63)=x$ is the exact solution. As a decimal, this is $x=5.97728$ (do this on your calculator).
\end{solution}

\part $6(10)^x-4=5$

\begin{solution}[5em]
Adding $4$ to both sides, then $6 \cdot 10^x = 9$. Dividing by $6$ yields $10^x = \frac{9}{6}=\frac{3}{2}$. Now, by the definition of the logarithm this means that $\log(3 /2) = x$. Recall that $\log $ with no subscript is base $10$. As a decimal, $x=0.17609$. 
\end{solution}

\end{parts}
