\begin{exercise}
A model rocket is launched vertically from the ground with an initial velocity of $168$ feet per second. Let $H(t)$ be the height, in feet, of the rocket $t$ seconds after launch.

    \begin{parts}
        \item Write a formula for the function $H(t)$. 

        \item Find $t$ when the model rocket hits the ground.

        \item Find the maximum height that the model rocket reaches.
    \end{parts}

\end{exercise}

\begin{solution}
    \begin{parts}
        \item $H(t)=-16t^2+168t$
        \item $10.5$ seconds
        \item The maximum height that the model rocket reaches is $441$ feet.
    \end{parts}
\end{solution}
