
In \exrange{exercise-4.2.1}{exercise-4.2.6}, solve an equation for $x$ or for $b$. Be sure to list all solutions. Give exact and approximate values rounded off to three decimal places. If there are no solutions say so.

\begin{exercise}\label{exercise-4.2.1}
$\displaystyle \frac{2}{5}x^4+3=11$
\end{exercise}

\begin{solution}
$x=-\sqrt[4]{20}\approx-2.115$ and $x=\sqrt[4]{20}\approx2.115$
\end{solution}

\begin{exercise}
$\displaystyle 1500b^{30}=17000$
\end{exercise}

\begin{solution}
$b=-\sqrt[30]{\frac{34}{3}} \approx -1.084$ and $b=\sqrt[30]{\frac{34}{3}} \approx 1.084$
\end{solution}

\begin{exercise}
$\displaystyle 5x^3-6=-8$
\end{exercise}

\begin{solution}
$x=\sqrt[3]{-\frac{2}{5}}\approx-0.737$
\end{solution}

\begin{exercise}
$\displaystyle \frac{5}{3}x^6+9=4$
\end{exercise}

\begin{solution}
No real solutions.
\end{solution}

\begin{exercise}
$\displaystyle 3(x-1)^2+1=10$
\end{exercise}

\begin{solution}
$x=1-\sqrt{3}\approx-0.732$ and $x=1+\sqrt{3}\approx2.732$
\end{solution}

\begin{exercise}\label{exercise-4.2.6}
$\displaystyle (x+2)^3=-8$
\end{exercise}

\begin{solution}
$x=-4$
\end{solution}

\noindent In \exrange{exercise-4.2.11}{exercise-4.2.14}, simplify a given expression. If it is not possible, say so. $x$ and $y$ are assumed to be positive.

\begin{exercise}\label{exercise-4.2.11}
$\displaystyle \frac{\sqrt{x}+\sqrt{y}}{xy}$
\end{exercise}

\begin{solution}
Cannot be simplified.
\end{solution}

\begin{exercise}
$\displaystyle (\sqrt[4]{y}\sqrt[3]{x})^4$
\end{exercise}

\begin{solution}
$xy\sqrt[3]{x}$
\end{solution}

\begin{exercise}
$\displaystyle \frac{x^{\frac{2}{3}}y^3}{xy}$
\end{exercise}

\begin{solution}
$\dfrac{y^2}{\sqrt[3]{x}}$
\end{solution}
