\noindent In \exrange{exercise-3.3.5}{exercise-3.3.8}, convert each quadratic function from vertex form to standard form and identify $a$, $b$, and $c$.


\begin{exercise}\label{exercise-3.3.5}
$f(x)=(x+1)^2+2$
\end{exercise}

\begin{solution}
    $f(x)=x^2+2x+3$; $a=1$, $b=2$, $c=3$
\end{solution}



\begin{exercise}
$g(x)=5(x-2)^2+6$
\end{exercise}

\begin{solution}
    $g(x)=5x^2-20x+26$; $a=5$, $b=-20$, $c=26$
\end{solution}



\begin{exercise}
$y=-2(t-4)^2+5$
\end{exercise}

\begin{solution}
    $y=-2t^2+16t-27$; $a=-2$, $b=16$, $c=-27$
\end{solution}



\begin{exercise}\label{exercise-3.3.8}
$h(x)=-(x+\frac{1}{2})^2+\frac{5}{3}$
\end{exercise}

\begin{solution}
    $h(x)=-x^2-x+\frac{17}{12}$; $a=-1$, $b=-1$, $c=\frac{17}{12}$
\end{solution}
