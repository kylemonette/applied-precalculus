            &lt;&lt;
                name:"a",
                snapWidth:0.25,
                point1: &lt;&lt; frozen: true &gt;&gt;,
                point2: &lt;&lt; frozen: true &gt;&gt;

        b = slider([-4.5,3.5], [-1.5,3.5], [-4,-1,4])
            &lt;&lt;
                name:"b",
                snapWidth:0.5,
                point1: &lt;&lt; frozen: true &gt;&gt;,
                point2: &lt;&lt; frozen: true &gt;&gt;
            &gt;&gt;;
        c = slider([-4.5,3], [-1.5,3], [-4,0,4])
            &lt;&lt;
                name:"c",
                snapWidth:0.5,
                point1: &lt;&lt; frozen: true &gt;&gt;,
                point2: &lt;&lt; frozen: true &gt;&gt;
            &gt;&gt;;
        fGraph = functiongraph(function(x) {
            return a.Value()*x^2 + b.Value()*x + c.Value();
        });
        P = glider(1,0, fGraph);
        Q = point(2,2)
            &lt;&lt; name:"P" &gt;&gt;;
        secant = line(P,Q)
            &lt;&lt; color: (function() {
                    m = (Q.Y() - P.Y())/(Q.X() - P.X());
                    dydx = 2*a.Value()*P.X() + b.Value();
                    cVal = "red";
                    if (abs(m-dydx) &lt; 0.001) {
                        cVal = "blue";
                    }
                    return cVal;
                })
            &gt;&gt;;
        dydxValue = text(-4, 1.5, function() {
                m = 2*a.Value()*P.X() + b.Value();
                return ("dy/dx = " + trunc(m,3));
            })
            &lt;&lt; frozen: true &gt;&gt;;
        mValue = text(-4, 1, function() {
                m = (Q.Y() - P.Y())/(Q.X() - P.X());
                return ("m = " + trunc(m,3));
            })
            &lt;&lt; frozen: true &gt;&gt;;
